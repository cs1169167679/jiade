var EDIT_PWD = '1011';   // 编辑密码
var DEL_PWD  = '1212';   // 删除密码
var LS_KEY   = 'lift_helper_v1';   // 数据存储的钥匙名

/* =========================================================
   全局数据
   ========================================================= */
var data = { brands: [], favorites: [], experience: [], debug: [] };
var unlocked = false;          // 是否已解锁
var currentBrandId = null;     // 当前打开的品牌

var SEC = {
  favorites:  { name: '收藏' },
  experience: { name: '经验' },
  debug:      { name: '调试资料' }
};

/* =========================================================
   小工具
   ========================================================= */
function $(id){ return document.getElementById(id); }

function uid(){
  return Date.now().toString(36) + Math.random().toString(36).slice(2, 7);
}

function toast(msg){
  var t = document.createElement('div');
  t.className = 'toast';
  t.textContent = msg;
  document.body.appendChild(t);
  setTimeout(function(){ t.remove(); }, 1800);
}

/* =========================================================
   数据保存 / 读取
   ========================================================= */
function saveData(){
  try {
    localStorage.setItem(LS_KEY, JSON.stringify(data));
  } catch(e) {
    toast('保存失败：存储空间可能已满');
  }
}

function loadData(){
  var s = null;
  try { s = localStorage.getItem(LS_KEY); } catch(e){}

  if(s){
    try {
      var d = JSON.parse(s);
      data.brands     = d.brands     || [];
      data.favorites  = d.favorites  || [];
      data.experience = d.experience || [];
      data.debug      = d.debug      || [];
    } catch(e){}
    return;
  }

  /* 第一次打开，放几条示例数据，不需要直接删掉即可 */
  data.brands = [{
    id: uid(),
    name: '示例品牌（可删除）',
    codes: [
      { id: uid(), code: 'E01', desc: '示例：门锁回路断开，检查门锁触点' },
      { id: uid(), code: 'E02', desc: '示例：变频器过流保护' }
    ]
  }];
  data.experience = [{
    id: uid(),
    title: '示例：电梯运行抖动排查',
    content: '1. 检查导靴磨损情况\n2. 检查钢丝绳张力是否均匀\n3. 检查导轨润滑\n4. 检查抱闸间隙'
  }];
  saveData();
}

/* =========================================================
   弹窗系统
   ========================================================= */
function openDialog(opt){
  $('dlgTitle').textContent = opt.title || '';
  var body = $('dlgBody');
  body.innerHTML = '';

  (opt.fields || []).forEach(function(f){
    var wrap = document.createElement('div');
    wrap.className = 'field';

    if(f.label){
      var lb = document.createElement('label');
      lb.textContent = f.label;
      wrap.appendChild(lb);
    }

    var el;
    if(f.type === 'textarea'){
      el = document.createElement('textarea');
      el.rows = f.rows || 4;
    } else {
      el = document.createElement('input');
      el.type = f.type || 'text';
      if(f.type === 'password') el.inputMode = 'numeric';
    }
    el.className = 'dlg-input';
    el.placeholder = f.placeholder || '';
    el.value = f.value || '';
    el.dataset.key = f.key;
    wrap.appendChild(el);
    body.appendChild(wrap);
  });

  var cancel = $('dlgCancel');
  cancel.style.display = '';
  cancel.onclick = closeDialog;

  var ok = $('dlgOk');
  ok.textContent = opt.okText || '确定';
  ok.className = 'btn-ok' + (opt.danger ? ' danger' : '');
  ok.onclick = function(){
    var vals = {};
    var inputs = body.querySelectorAll('[data-key]');
    for(var i = 0; i < inputs.length; i++){
      vals[inputs[i].dataset.key] = inputs[i].value.trim();
    }
    var r = opt.onOk ? opt.onOk(vals) : undefined;
    if(r !== false) closeDialog();
  };

  $('mask').style.display = 'flex';
  setTimeout(function(){
    var first = body.querySelector('input, textarea');
    if(first) first.focus();
  }, 100);
}

function closeDialog(){
  $('mask').style.display = 'none';
}

/* 只读详情弹窗 */
function openDetail(title, content){
  $('dlgTitle').textContent = title;
  var body = $('dlgBody');
  body.innerHTML = '';

  var d = document.createElement('div');
  d.className = 'detail-text';
  d.textContent = content || '（无内容）';
  body.appendChild(d);

  $('dlgCancel').style.display = 'none';

  var ok = $('dlgOk');
  ok.textContent = '关闭';
  ok.className = 'btn-ok';
  ok.onclick = closeDialog;

  $('mask').style.display = 'flex';
}

/* =========================================================
   锁：编辑密码 1011
   ========================================================= */
function updateLockUI(){
  var btn = $('lockBtn');
  btn.textContent = unlocked ? '🔓' : '🔒';
  btn.className = 'lock-btn' + (unlocked ? ' on' : '');
  document.body.classList.toggle('unlocked', unlocked);

  var sb = $('statusBar');
  if(unlocked){
    sb.textContent = '🔓 已解锁，可以编辑';
    sb.classList.add('on');
  } else {
    sb.textContent = '🔒 只读模式（点右上角 🔒 输入密码解锁）';
    sb.classList.remove('on');
  }
}

function requireUnlock(cb){
  if(unlocked){ cb(); return; }

  openDialog({
    title: '🔒 请输入编辑密码',
    fields: [{ key: 'pwd', type: 'password', placeholder: '编辑密码' }],
    okText: '解锁',
    onOk: function(v){
      if(v.pwd !== EDIT_PWD){
        toast('密码错误');
        return false;
      }
      unlocked = true;
      updateLockUI();
      toast('已解锁，可以编辑');
      closeDialog();
      setTimeout(cb, 60);
    }
  });
}

function onLockClick(){
  if(unlocked){
    unlocked = false;
    updateLockUI();
    toast('已锁定，只能查看');
  } else {
    requireUnlock(function(){});
  }
}

/* =========================================================
   删除：二次确认 + 删除密码 1212
   ========================================================= */
function askDelete(name, cb){
  openDialog({
    title: '⚠️ 删除确认',
    fields: [{
      key: 'pwd',
      label: '确定要删除 ' + name + ' 吗？\n删除后无法恢复！\n\n请输入删除密码：',
      type: 'password',
      placeholder: '删除密码'
    }],
    okText: '确认删除',
    danger: true,
    onOk: function(v){
      if(v.pwd !== DEL_PWD){
        toast('删除密码错误');
        return false;
      }
      cb();
      toast('已删除');
    }
  });
}

   新增：品牌 / 故障码 / 资料

function addBrand(){
  requireUnlock(function(){
    openDialog({
      title: '添加品牌',
      fields: [{ key: 'name', label: '品牌名称', placeholder: '例如：日立 / 三菱 / 通力' }],
      onOk: function(v){
        if(!v.name){ toast('请输入品牌名称'); return false; }
        data.brands.push({ id: uid(), name: v.name, codes: [] });
        saveData();
        renderHome();
        toast('添加成功');
      }
    });
  });
}

function addCode(){
  var brand = getBrand();
  if(!brand) return;

  requireUnlock(function(){
    openDialog({
      title: '添加故障码',
      fields: [
        { key: 'code', label: '故障码', placeholder: '例如：E01' },
        { key: 'desc', label: '故障定义 / 原因 / 处理', type: 'textarea', rows: 4,
          placeholder: '例如：门锁回路断开，检查门锁触点' }
      ],
      onOk: function(v){
        if(!v.code){ toast('请输入故障码'); return false; }
        brand.codes.push({ id: uid(), code: v.code, desc: v.desc });
        saveData();
        renderBrand();
        toast('添加成功');
      }
    });
  });
}

function addItem(type){
  requireUnlock(function(){
    openDialog({
      title: '添加' + SEC[type].name,
      fields: [
        { key: 'title',   label: '标题', placeholder: '请输入标题' },
        { key: 'content', label: '内容', type: 'textarea', rows: 5, placeholder: '请输入内容' }
      ],
      onOk: function(v){
        if(!v.title){ toast('请输入标题'); return false; }
        data[type].push({ id: uid(), title: v.title, content: v.content });
        saveData();
        renderHome();
        toast('添加成功');
      }
    });
  });
}

function getBrand(){
  for(var i = 0; i < data.brands.length; i++){
    if(data.brands[i].id === currentBrandId) return data.brands[i];
  }
  return null;
}

/* =========================================================
   卡片工厂
   ========================================================= */
function makeCard(title, sub, onClick, onDel){
  var row = document.createElement('div');
  row.className = 'card row';

  var main = document.createElement('div');
  main.className = 'card-main';
  main.onclick = onClick;

  var t = document.createElement('div');
  t.className = 'card-title';
  t.textContent = title;
  main.appendChild(t);

  if(sub){
    var s = document.createElement('div');
    s.className = 'card-sub';
    s.textContent = sub;
    main.appendChild(s);
  }

  var del = document.createElement('button');
  del.className = 'del-btn';
  del.textContent = '删除';
  del.onclick = function(e){
    e.stopPropagation();
    onDel();
  };

  row.appendChild(main);
  row.appendChild(del);
  return row;
}

/* =========================================================
   渲染主页
   ========================================================= */
function renderHome(){
  var kw = ($('searchInput').value || '').trim().toLowerCase();

  /* ---- 品牌 ---- */
  var bl = $('brandList');
  bl.innerHTML = '';

  var brands = data.brands.filter(function(b){
    if(!kw) return true;
    if((b.name || '').toLowerCase().indexOf(kw) >= 0) return true;
    return (b.codes || []).some(function(c){
      return ((c.code || '') + (c.desc || '')).toLowerCase().indexOf(kw) >= 0;
    });
  });

  if(brands.length === 0){
    bl.innerHTML = '<div class="empty">' +
      (kw ? '没有找到相关品牌' : '暂无品牌，点右上角添加') + '</div>';
  } else {
    brands.forEach(function(b){
      bl.appendChild(makeCard(
        b.name,
        (b.codes || []).length + ' 条故障码',
        function(){ openBrand(b.id); },
        function(){
          askDelete('品牌「' + b.name + '」', function(){
            data.brands = data.brands.filter(function(x){ return x.id !== b.id; });
            saveData();
            renderHome();
          });
        }
      ));
    });
  }

  /* ---- 收藏 / 经验 / 调试资料 ---- */
  ['favorites', 'experience', 'debug'].forEach(function(type){
    var box = $(type + 'List');
    box.innerHTML = '';

    var arr = (data[type] || []).filter(function(it){
      if(!kw) return true;
      return ((it.title || '') + (it.content || '')).toLowerCase().indexOf(kw) >= 0;
    });

    if(arr.length === 0){
      box.innerHTML = '<div class="empty">' +
        (kw ? '没有找到相关内容' : '暂无内容，点右上角添加') + '</div>';
      return;
    }

    arr.forEach(function(it){
      var sub = (it.content || '').replace(/\n/g, ' ');
      if(sub.length > 30) sub = sub.slice(0, 30) + '…';

      box.appendChild(makeCard(
        it.title,
        sub,
        function(){ openDetail(it.title, it.content); },
        function(){
          askDelete(SEC[type].name + '「' + it.title + '」', function(){
            data[type] = data[type].filter(function(x){ return x.id !== it.id; });
            saveData();
            renderHome();
          });
        }
      ));
    });
  });
}

/* =========================================================
   渲染品牌详情
   ========================================================= */
function renderBrand(){
  var brand = getBrand();
  if(!brand){ goHome(); return; }

  $('brandTitle').textContent = brand.name;

  var box = $('codeList');
  box.innerHTML = '';

  if(!brand.codes || brand.codes.length === 0){
    box.innerHTML = '<div class="empty">暂无故障码，点右上角添加</div>';
    return;
  }

  brand.codes.forEach(function(c){
    var row = document.createElement('div');
    row.className = 'card row';

    var main = document.createElement('div');
    main.className = 'card-main';

    var codeEl = document.createElement('div');
    codeEl.className = 'code';
    codeEl.textContent = c.code;
    main.appendChild(codeEl);

    if(c.desc){
      var descEl = document.createElement('div');
      descEl.className = 'desc';
      descEl.textContent = c.desc;
      main.appendChild(descEl);
    }

    var del = document.createElement('button');
    del.className = 'del-btn';
    del.textContent = '删除';
    del.onclick = function(){
      askDelete('故障码 ' + c.code, function(){
        brand.codes = brand.codes.filter(function(x){ return x.id !== c.id; });
        saveData();
        renderBrand();
      });
    };

    row.appendChild(main);
    row.appendChild(del);
    box.appendChild(row);
  });
}

/* =========================================================
   页面跳转
   ========================================================= */
function openBrand(id){
  currentBrandId = id;
  $('page-home').style.display = 'none';
  $('page-brand').style.display = 'block';
  window.scrollTo(0, 0);
  renderBrand();
}

function goHome(){
  currentBrandId = null;
  $('page-home').style.display = 'block';
  $('page-brand').style.display = 'none';
  window.scrollTo(0, 0);
  renderHome();
}
