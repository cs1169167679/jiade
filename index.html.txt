function toast(msg){
  const t = document.createElement('div');
  t.className = 'toast';
  t.textContent = msg;
  document.body.appendChild(t);
  requestAnimationFrame(() => t.classList.add('show'));

function showSheet(opts){
  const overlay = document.createElement('div');
  overlay.className = 'overlay';
  overlay.innerHTML = `
    <div class="sheet">
      <div class="sheet-title">${opts.title || ''}</div>
      ${opts.desc ? `<div class="sheet-desc">${opts.desc}</div>` : ''}
      <div class="sheet-body"></div>
      <div class="sheet-actions"></div>
    </div>`;
  const bodyEl = overlay.querySelector('.sheet-body');
  if(typeof opts.body === 'string') bodyEl.innerHTML = opts.body;
  else if(opts.body) bodyEl.appendChild(opts.body);

  const actions = overlay.querySelector('.sheet-actions');
  (opts.buttons || []).forEach(b => {
    const btn = document.createElement('button');
    btn.className = 'btn ' + (b.primary ? 'primary' : 'ghost') + (b.danger ? ' danger' : '');
    btn.textContent = b.text;
    btn.addEventListener('click', () => {
      if(b.onClick) b.onClick(overlay);
      if(!b.keepOpen) close();
    });
    actions.appendChild(btn);
  });

  document.body.appendChild(overlay);
  overlay._close = close;
  function close(){ if(overlay.parentNode) overlay.remove(); }
  if(opts.onMount) opts.onMount(overlay, close);
  return overlay;
}

function confirmDialog(title, desc, okText='继续删除'){
  return new Promise(resolve => {
    showSheet({
      title, desc,
      buttons: [
        {text:'取消', onClick: () => resolve(false)},
        {text: okText, danger:true, onClick: () => resolve(true)}
      ]
    });
  });
}

          if(v === DB.password){
            o._close();
            resolve(true);

            const inp = o.querySelector('.pw');

        i.addEventListener('keydown', e => {
          if(e.key === 'Enter') o.querySelector('.btn.primary').click();
        });
      }
    });
  });

  if(cHits.length){
    blocks.push(`<div class="section-head"><h2>故障码</h2><span class="count">${cHits.length}</span></div>`);
    blocks.push(cHits.map(x => codeCardHTML(x.b.id, x.it, x.b.name)).join(''));
  }

  const eHits = DB.experiences.filter(x => has(x.title) || has(x.content));
  if(eHits.length){
    blocks.push(`<div class="section-head"><h2>经验</h2><span class="count">${eHits.length}</span></div>`);
    blocks.push(eHits.map(i => itemCardHTML('experiences', i)).join(''));
  }

  const dHits = DB.debugDocs.filter(x => has(x.title) || has(x.content));
  if(dHits.length){
    blocks.push(`<div class="section-head"><h2>调试资料</h2><span class="count">${dHits.length}</span></div>`);
    blocks.push(dHits.map(i => itemCardHTML('debugDocs', i)).join(''));
  }

  if(!blocks.length){
    return `<div class="empty">没有找到「${esc(q)}」相关内容</div>`;
  }
  return blocks.join('');
}

function doExport(){
  const payload = {
    app: '电梯维修助手',
    version: 1,
    exportedAt: new Date().toISOString(),
    data: DB
  };
  const text = JSON.stringify(payload, null, 2);
  const d = new Date();
  const stamp = `${d.getFullYear()}${String(d.getMonth()+1).padStart(2,'0')}${String(d.getDate()).padStart(2,'0')}`;
  const filename = `电梯维修助手_备份_${stamp}.json`;

  try{
    const blob = new Blob([text], {type:'application/json;charset=utf-8'});
    const url = URL.createObjectURL(blob);
    const a = document.createElement('a');
    a.href = url;
    a.download = filename;
    document.body.appendChild(a);
    a.click();
    a.remove();
    setTimeout(() => URL.revokeObjectURL(url), 2000);
    toast('已导出：' + filename);
  }catch(e){
    toast('导出失败，请检查浏览器权限');
  }
}

function doImport(){
  const input = document.createElement('input');
  input.type = 'file';
  input.accept = '.json,application/json';
  input.style.display = 'none';
  document.body.appendChild(input);

  input.addEventListener('change', () => {
    const file = input.files && input.files[0];
    input.remove();
    if(!file) return;

    const reader = new FileReader();
    reader.onload = async () => {
      let parsed;
      try{ parsed = JSON.parse(reader.result); }
      catch(e){ toast('文件格式错误，无法解析'); return; }

      const incoming = parsed && parsed.data ? parsed.data : parsed;
      if(!incoming || typeof incoming !== 'object' ||
         (!Array.isArray(incoming.brands) && !Array.isArray(incoming.experiences) && !Array.isArray(incoming.debugDocs))){
        toast('不是有效的备份文件');
        return;
      }

      const nB = (incoming.brands || []).length;
      const nE = (incoming.experiences || []).length;
      const nD = (incoming.debugDocs || []).length;

      const ok = await askPassword({
        title:'导入数据',
        desc:'导入会修改本地数据，请输入管理密码'
      });
      if(!ok) return;

      showSheet({
        title:'选择导入方式',
        desc:`文件包含：<b>${nB}</b> 个品牌 / <b>${nE}</b> 条经验 / <b>${nD}</b> 条调试资料`,
        buttons:[
          {text:'取消'},
          {text:'合并导入', onClick: () => applyImport(incoming, 'merge')},
          {text:'覆盖导入', primary:true, danger:true, onClick: () => applyImport(incoming, 'replace')}
        ]
      });
    };
    reader.readAsText(file, 'utf-8');
  });

  input.click();
}

function applyImport(incoming, mode){
  if(mode === 'replace'){
    DB = {
      password   : incoming.password || DB.password,
      brands     : Array.isArray(incoming.brands) ? incoming.brands : [],
      experiences: Array.isArray(incoming.experiences) ? incoming.experiences : [],
      debugDocs  : Array.isArray(incoming.debugDocs) ? incoming.debugDocs : []
    };
  }else{
    (incoming.brands || []).forEach(nb => {
      if(!nb || !nb.name) return;
      const exist = DB.brands.find(b => b.name === nb.name);
      if(!exist){
        DB.brands.push({
          id: nb.id || uid(),
          name: nb.name,
          note: nb.note || '',
          items: Array.isArray(nb.items) ? nb.items : []
        });
      }else{
        const seen = new Set(exist.items.map(i => (i.code || '') + '|' + (i.desc || '')));
        (nb.items || []).forEach(it => {
          const key = (it.code || '') + '|' + (it.desc || '');
          if(!seen.has(key)){
            seen.add(key);
            exist.items.push({id: it.id || uid(), code: it.code || '', desc: it.desc || '', fav: !!it.fav});
          }
        });
      }
    });
    [['experiences','experiences'], ['debugDocs','debugDocs']].forEach(([k]) => {
      DB[k] = DB[k] || [];
      const seen = new Set(DB[k].map(i => (i.title || '') + '|' + (i.content || '')));
      (incoming[k] || []).forEach(it => {
        const key = (it.title || '') + '|' + (it.content || '');
        if(!seen.has(key)){
          seen.add(key);
          DB[k].push({id: it.id || uid(), title: it.title || '', content: it.content || '', fav: !!it.fav});
        }
      });
    });
  }
  saveDB();
  state.query = '';
  state.brandQuery = '';
  render();
  toast(mode === 'replace' ? '覆盖导入完成' : '合并导入完成');
}

function initHoldToUnlock(){
  const title = $('#appTitle');
  const bar   = $('#holdBar');
  const fill  = bar.querySelector('i');

  let startAt = 0, rafId = null, sx = 0, sy = 0;

  function cancel(){
    if(!startAt) return;
    startAt = 0;
    cancelAnimationFrame(rafId);
    bar.classList.remove('show');
    fill.style.width = '0%';
  }

  function finish(){
    startAt = 0;
    cancelAnimationFrame(rafId);
    bar.classList.remove('show');
    fill.style.width = '0%';
    if(navigator.vibrate) navigator.vibrate(30);
    openLockPanel();
  }

  function begin(e){
    if(startAt) return;
    startAt = Date.now();
    sx = e.clientX; sy = e.clientY;
    bar.classList.add('show');

    const tick = () => {
      if(!startAt) return;
      const p = Math.min(1, (Date.now() - startAt) / HOLD_MS);
      fill.style.width = (p * 100) + '%';
      if(p >= 1){ finish(); return; }
      rafId = requestAnimationFrame(tick);
    };
    rafId = requestAnimationFrame(tick);
  }

  title.addEventListener('pointerdown', e => {
    e.preventDefault();
    begin(e);
  });
  title.addEventListener('pointerup', cancel);
  title.addEventListener('pointercancel', cancel);
  title.addEventListener('pointermove', e => {
    if(!startAt) return;
    if(Math.hypot(e.clientX - sx, e.clientY - sy) > 14) cancel();
  });

  if(!window.PointerEvent){
    title.addEventListener('touchstart', e => begin(e.touches[0]), {passive:true});
    title.addEventListener('touchend', cancel);
    title.addEventListener('touchcancel', cancel);
    title.addEventListener('touchmove', e => {
      if(!startAt) return;
      const t = e.touches[0];
      if(Math.hypot(t.clientX - sx, t.clientY - sy) > 14) cancel();
    }, {passive:true});
  }
}

        const err = o.querySelector('.err');
        if(oldPw !== DB.password){ err.textContent = '当前密码错误'; return; }
        if(n1.length < 4 || n1.length > 12){ err.textContent = '新密码需 4-12 位'; return; }
        if(n1 !== n2){ err.textContent = '两次输入不一致'; return; }
        DB.password = n1;
        saveDB();
        o._close();
        toast('密码已修改，请牢记');
      }}
    ]
  });
}

function handleClick(e){
  const t = e.target.closest('[data-act]');
  if(!t) return;
  const act = t.dataset.act;
  const id  = t.dataset.id;

  switch(act){
    case 'nav':
      state.view = t.dataset.view;
      state.query = '';
      state.brandQuery = '';
      render();
      break;

    case 'back':
      state.view = 'home';
      state.brandId = null;
      state.brandQuery = '';
      render();
      break;

    case 'open-brand':
      state.view = 'brand';
      state.brandId = id;
      state.brandQuery = '';
      render();
      break;

    case 'add-brand':  addBrand(); break;
    case 'edit-brand': editBrand(id); break;
    case 'del-brand':  delBrand(id); break;

    case 'add-code':  addCode(); break;
    case 'edit-code': editCode(id, t.dataset.parent); break;
    case 'del-code':  delCode(id, t.dataset.parent); break;

    case 'add-item':  addItem(t.dataset.type); break;
    case 'edit-item': editItem(t.dataset.type, id); break;
    case 'del-item':  delItem(t.dataset.type, id); break;

    case 'toggle-fav': {
      const on = toggleFav(t.dataset.type, id, t.dataset.parent);
      if(state.view === 'favorites'){
        render({keepScroll:true});
      }else{
        t.classList.toggle('on', on);
        t.textContent = on ? '★' : '☆';
      }
      break;
    }

    case 'export': doExport(); break;
    case 'import': doImport(); break;
  }
}

(function init(){
  $('#view').addEventListener('click', handleClick);
  initHoldToUnlock();
  render();

  let lastTouch = 0;
  document.addEventListener('touchend', e => {
    const now = Date.now();
    if(now - lastTouch <= 300) e.preventDefault();
    lastTouch = now;
  }, {passive:false});
})();
</script>
